// ==================================================
// VELOCIDADES
// ==================================================

// X/Y suaves
int VELOCIDADE_XY = 105;

// Z SUBIDA
int VELOCIDADE_Z_SUBIDA = 255;


// ==================================================
// FORÇA INICIAL
// ==================================================

int KICK_XY = 190;

// Z SUBIDA
int KICK_Z_SUBIDA = 255;


// ==================================================
// DESLOCAMENTOS
// ==================================================

int DESLOCAMENTO_X = 180;
int DESLOCAMENTO_Y = 220;
int DESLOCAMENTO_Z = 500;


// ==================================================
// EIXO X
// ==================================================

int ENX = 9;

int X1  = 2;
int X2  = 3;


// ==================================================
// EIXO Y
// ==================================================

int ENY = 10;

int Y1  = 4;
int Y2  = 5;


// ==================================================
// EIXO Z MOTOR 1
// ==================================================

int ENZ1 = 11;

int Z1A  = 6;
int Z1B  = 7;


// ==================================================
// EIXO Z MOTOR 2
// ==================================================

int ENZ2 = 12;

int Z2A  = 8;
int Z2B  = 13;



// ==================================================
void setup() {

  // X
  pinMode(ENX, OUTPUT);
  pinMode(X1, OUTPUT);
  pinMode(X2, OUTPUT);

  // Y
  pinMode(ENY, OUTPUT);
  pinMode(Y1, OUTPUT);
  pinMode(Y2, OUTPUT);

  // Z1
  pinMode(ENZ1, OUTPUT);
  pinMode(Z1A, OUTPUT);
  pinMode(Z1B, OUTPUT);

  // Z2
  pinMode(ENZ2, OUTPUT);
  pinMode(Z2A, OUTPUT);
  pinMode(Z2B, OUTPUT);

  pararTudo();
}



// ==================================================
void loop() {

  // =========================================
  // EIXO X
  // =========================================

  moverXY(ENX, X1, X2, true, DESLOCAMENTO_X);

  delay(250);

  moverXY(ENX, X1, X2, false, DESLOCAMENTO_X);

  delay(500);


  // =========================================
  // EIXO Y
  // =========================================

  moverXY(ENY, Y1, Y2, true, DESLOCAMENTO_Y);

  delay(250);

  moverXY(ENY, Y1, Y2, false, DESLOCAMENTO_Y);

  delay(500);


  // =========================================
  // EIXO Z SOBE
  // =========================================

  moverZSubida(DESLOCAMENTO_Z);

  delay(400);


  // =========================================
  // EIXO Z DESCE FREANDO
  // =========================================

  moverZDescida(DESLOCAMENTO_Z);

  delay(1500);
}



// ==================================================
// MOVIMENTO X/Y
// ==================================================
void moverXY(int EN, int P1, int P2, bool sentido, int tempoTotal) {

  // DIREÇÃO
  if (sentido) {

    digitalWrite(P1, HIGH);
    digitalWrite(P2, LOW);

  } else {

    digitalWrite(P1, LOW);
    digitalWrite(P2, HIGH);
  }

  // KICK
  analogWrite(EN, KICK_XY);

  delay(60);


  // ACELERAÇÃO
  for (int v = KICK_XY; v >= VELOCIDADE_XY; v -= 10) {

    analogWrite(EN, v);

    delay(8);
  }


  // MOVIMENTO
  analogWrite(EN, VELOCIDADE_XY);

  delay(tempoTotal);


  // FRENAGEM
  for (int v = VELOCIDADE_XY; v >= 0; v -= 10) {

    analogWrite(EN, v);

    delay(8);
  }


  // PARADA
  analogWrite(EN, 0);

  digitalWrite(P1, LOW);
  digitalWrite(P2, LOW);
}



// ==================================================
// Z SUBIDA FORTE
// ==================================================
void moverZSubida(int tempoTotal) {

  // DIREÇÃO SUBIDA
  digitalWrite(Z1A, HIGH);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, HIGH);
  digitalWrite(Z2B, LOW);


  // FORÇA TOTAL
  analogWrite(ENZ1, KICK_Z_SUBIDA);
  analogWrite(ENZ2, KICK_Z_SUBIDA);

  delay(120);


  // MOVIMENTO
  analogWrite(ENZ1, VELOCIDADE_Z_SUBIDA);
  analogWrite(ENZ2, VELOCIDADE_Z_SUBIDA);

  delay(tempoTotal);


  // DESACELERAÇÃO SUAVE
  for (int v = VELOCIDADE_Z_SUBIDA; v >= 120; v -= 10) {

    analogWrite(ENZ1, v);
    analogWrite(ENZ2, v);

    delay(8);
  }


  // PARADA
  analogWrite(ENZ1, 0);
  analogWrite(ENZ2, 0);

  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, LOW);
}



// ==================================================
// Z DESCIDA FREANDO
// NÃO ACELERA
// APENAS CONTROLA A GRAVIDADE
// ==================================================
void moverZDescida(int tempoTotal) {

  // DIREÇÃO DESCIDA
  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, HIGH);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, HIGH);


  // =========================================
  // FREIO INICIAL
  // =========================================

  int pwm = 85;

  analogWrite(ENZ1, pwm);
  analogWrite(ENZ2, pwm);


  // =========================================
  // DESCIDA CONTROLADA
  // =========================================

  for (int t = 0; t < tempoTotal; t += 25) {

    analogWrite(ENZ1, pwm);
    analogWrite(ENZ2, pwm);

    delay(25);


    // REDUZ MUITO LENTAMENTE
    if (pwm > 45 && t % 120 == 0) {

      pwm--;
    }
  }


  // =========================================
  // FRENAGEM FINAL SUAVE
  // =========================================

  for (int v = pwm; v >= 30; v--) {

    analogWrite(ENZ1, v);
    analogWrite(ENZ2, v);

    delay(35);
  }


  // =========================================
  // ENCOSTE FINAL SUAVE
  // =========================================

  analogWrite(ENZ1, 22);
  analogWrite(ENZ2, 22);

  delay(60);


  // =========================================
  // DESLIGA RÁPIDO APÓS ENCOSTAR
  // =========================================

  analogWrite(ENZ1, 0);
  analogWrite(ENZ2, 0);

  delay(20);

  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, LOW);
}



// ==================================================
void pararTudo() {

  analogWrite(ENX, 0);
  analogWrite(ENY, 0);

  analogWrite(ENZ1, 0);
  analogWrite(ENZ2, 0);

  digitalWrite(X1, LOW);
  digitalWrite(X2, LOW);

  digitalWrite(Y1, LOW);
  digitalWrite(Y2, LOW);

  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, LOW);
}
