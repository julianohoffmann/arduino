// ==================================================
// VELOCIDADES
// ==================================================

int VELOCIDADE_XY = 95;

int KICK_XY = 180;

int VELOCIDADE_Z = 255;


// ==================================================
// MOVIMENTOS
// ==================================================

int MICRO = 45;
int CURTO = 80;
int MEDIO = 120;

int ALTURA_Z = 220;


// ==================================================
// EIXO X
// ==================================================

int ENX = 9;

int X1 = 2;
int X2 = 3;


// ==================================================
// EIXO Y
// ==================================================

int ENY = 10;

int Y1 = 4;
int Y2 = 5;


// ==================================================
// Z MOTOR 1
// ==================================================

int ENZ1 = 11;

int Z1A = 6;
int Z1B = 7;


// ==================================================
// Z MOTOR 2
// ==================================================

int ENZ2 = 12;

int Z2A = 8;
int Z2B = 13;



// ==================================================
void setup() {

  pinMode(ENX, OUTPUT);
  pinMode(X1, OUTPUT);
  pinMode(X2, OUTPUT);

  pinMode(ENY, OUTPUT);
  pinMode(Y1, OUTPUT);
  pinMode(Y2, OUTPUT);

  pinMode(ENZ1, OUTPUT);
  pinMode(Z1A, OUTPUT);
  pinMode(Z1B, OUTPUT);

  pinMode(ENZ2, OUTPUT);
  pinMode(Z2A, OUTPUT);
  pinMode(Z2B, OUTPUT);

  pararTudo();
}



// ==================================================
void loop() {

  // =========================================
  // ELEVAÇÃO SUAVE
  // =========================================

  subirZ();

  delay(400);


  // =========================================
  // ESPIRAL SUAVE
  // =========================================

  espiral();

  delay(250);


  // =========================================
  // ÓRBITA
  // =========================================

  orbita();

  delay(250);


  // =========================================
  // ASSINATURA FINAL
  // =========================================

  assinatura();

  delay(500);


  // =========================================
  // DESCIDA AMORTECIDA
  // =========================================

  descerZ();

  delay(2500);
}



// ==================================================
// ESPIRAL
// ==================================================
void espiral() {

  moverX(true, MICRO);
  moverY(true, MICRO);

  moverX(true, CURTO);
  moverY(true, MICRO);

  moverX(false, CURTO);
  moverY(true, CURTO);

  moverX(false, MEDIO);
  moverY(false, CURTO);

  moverX(true, MICRO);
  moverY(false, MEDIO);
}



// ==================================================
// ÓRBITA
// ==================================================
void orbita() {

  for (int i = 0; i < 2; i++) {

    moverX(true, CURTO);

    moverY(true, MICRO);

    moverX(false, MICRO);

    moverY(true, CURTO);

    moverX(false, CURTO);

    moverY(false, MICRO);

    moverX(true, MICRO);

    moverY(false, CURTO);
  }
}



// ==================================================
// ASSINATURA FINAL
// ==================================================
void assinatura() {

  moverX(true, MEDIO);

  moverY(true, MICRO);

  moverX(false, CURTO);

  moverY(false, MICRO);

  moverX(true, MICRO);

  moverY(true, CURTO);

  moverX(false, MICRO);

  moverY(false, CURTO);
}



// ==================================================
// MOVIMENTO X
// ==================================================
void moverX(bool sentido, int tempoTotal) {

  moverXY(ENX, X1, X2, sentido, tempoTotal);
}



// ==================================================
// MOVIMENTO Y
// ==================================================
void moverY(bool sentido, int tempoTotal) {

  moverXY(ENY, Y1, Y2, sentido, tempoTotal);
}



// ==================================================
// MOVIMENTO XY SUAVE
// ==================================================
void moverXY(int EN, int P1, int P2, bool sentido, int tempoTotal) {

  // DIREÇÃO
  if (sentido) {

    digitalWrite(P1, HIGH);
    digitalWrite(P2, LOW);

  } else {

    digitalWrite(P1, LOW);
    digitalWrite(P2, HIGH);
  }


  // ARRANQUE SUAVE
  analogWrite(EN, KICK_XY);

  delay(20);


  // TRANSIÇÃO
  for (int v = KICK_XY; v >= VELOCIDADE_XY; v -= 8) {

    analogWrite(EN, v);

    delay(4);
  }


  // MOVIMENTO
  analogWrite(EN, VELOCIDADE_XY);

  delay(tempoTotal);


  // DESACELERAÇÃO
  for (int v = VELOCIDADE_XY; v >= 0; v -= 6) {

    analogWrite(EN, v);

    delay(5);
  }


  // PARADA
  analogWrite(EN, 0);

  digitalWrite(P1, LOW);
  digitalWrite(P2, LOW);

  delay(20);
}



// ==================================================
// SUBIDA Z
// ==================================================
void subirZ() {

  digitalWrite(Z1A, HIGH);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, HIGH);
  digitalWrite(Z2B, LOW);

  analogWrite(ENZ1, VELOCIDADE_Z);
  analogWrite(ENZ2, VELOCIDADE_Z);

  delay(ALTURA_Z);


  // AMORTECIMENTO
  for (int v = VELOCIDADE_Z; v >= 100; v -= 10) {

    analogWrite(ENZ1, v);
    analogWrite(ENZ2, v);

    delay(6);
  }

  analogWrite(ENZ1, 0);
  analogWrite(ENZ2, 0);

  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, LOW);
}



// ==================================================
// DESCIDA Z AMORTECIDA
// ==================================================
void descerZ() {

  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, HIGH);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, HIGH);


  int pwm = 80;

  analogWrite(ENZ1, pwm);
  analogWrite(ENZ2, pwm);


  // DESCIDA CONTROLADA
  for (int t = 0; t < ALTURA_Z; t += 20) {

    analogWrite(ENZ1, pwm);
    analogWrite(ENZ2, pwm);

    delay(20);

    // reduz lentamente
    if (pwm > 42 && t % 100 == 0) {

      pwm--;
    }
  }


  // TOQUE FINAL
  for (int v = pwm; v >= 20; v--) {

    analogWrite(ENZ1, v);
    analogWrite(ENZ2, v);

    delay(30);
  }


  analogWrite(ENZ1, 0);
  analogWrite(ENZ2, 0);

  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, LOW);
}



// ==================================================
void pararTudo() {

  analogWrite(ENX, 0);
  analogWrite(ENY, 0);

  analogWrite(ENZ1, 0);
  analogWrite(ENZ2, 0);

  digitalWrite(X1, LOW);
  digitalWrite(X2, LOW);

  digitalWrite(Y1, LOW);
  digitalWrite(Y2, LOW);

  digitalWrite(Z1A, LOW);
  digitalWrite(Z1B, LOW);

  digitalWrite(Z2A, LOW);
  digitalWrite(Z2B, LOW);
}
